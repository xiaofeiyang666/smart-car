  86: /* ===================== Ostu ===================== */
  87: uint8 Ostu(void)
  88: {
  89:     uint16 hist[256] = {0};
  90:     uint32 total, sum = 0, sumB = 0, wB = 0, wF = 0;
  91:     float mB, mF, diff, between, maxBetween = -1.0f;
  92:     uint8 threshold = 0;
  93:     uint16 i, x, y;
  94: 
  95:     for(y = 0; y < MT9V03X_H; y += 2) {
  96:         for(x = 0; x < MT9V03X_W; x += 2) {
  97:             hist[mt9v03x_image[y][x]]++;
  98:         }
  99:     }
 100:     total = (MT9V03X_H / 2) * (MT9V03X_W / 2);
 101: 
 102:     for(i = 0; i < 256; i++) sum += (uint32)i * hist[i];
 103: 
 104:     for(i = 0; i < 256; i++) {
 105:         wB += hist[i];
 106:         if(wB == 0) continue;
 107:         wF = total - wB;
 108:         if(wF == 0) break;
 109: 
 110:         sumB += (uint32)i * hist[i];
 111:         mB = (float)sumB / wB;
 112:         mF = (float)(sum - sumB) / wF;
 113: 
 114:         diff = mB - mF;
 115:         between = (float)wB * (float)wF * diff * diff;
 116: 
 117:         if(between > maxBetween) {
 118:             maxBetween = between;
 119:             threshold = (uint8)i;
 120:         }
 121:     }
 122:     return threshold;
 123: }

 205: /* =======================================================================
 206:  * 八邻域边界跟踪算法 (8-Neighborhood Edge Tracking) - 防竖线修复版
 207:  * ======================================================================= */
 208: void image_deal(void)
 209: {
 210:     /* --- 1. 变量声明 (C89标准) --- */
 211:     int i;
 212:     int start_left_x, start_right_x, start_y;
 213:     int curr_x, curr_y, curr_dir;
 214:     int search_dir, found, start_search;
 215:     int nx, ny;
 216: 
 217:     /* --- 2. 初始化数组 --- */
 218:     for (i = 0; i < MT9V03X_H; i++) {
 219:         left_line_list[i] = 255;
 220:         right_line_list[i] = 255;
 221:         mid_line_list[i] = MID_W;
 222:     }
 223:     left_pts_cnt = 0;
 224:     right_pts_cnt = 0;
 225: 
 226:     /* --- 3. 获取起点 --- */
 227:     start_left_x = left_jidian;
 228:     start_right_x = right_jidian;
 229:     start_y = jidian_search_line - 1;
 230:     
 231:     if (start_left_x < 1) start_left_x = 1;
 232:     if (start_right_x > MT9V03X_W - 2) start_right_x = MT9V03X_W - 2;
 233: 
 234:     /* ---------------- 左边缘八邻域爬行 ---------------- */
 235:     curr_x = start_left_x;
 236:     curr_y = start_y;
 237:     curr_dir = 0; /* 初始方向：向上 */
 238: 
 239:     while (left_pts_cnt < MT9V03X_H * 3) {
 240:         /* 记录点集供高级算法使用 */
 241:         left_pts_x[left_pts_cnt] = curr_x;
 242:         left_pts_y[left_pts_cnt] = curr_y;
 243:         left_pts_dir[left_pts_cnt] = curr_dir;
 244:         left_pts_cnt++;
 245: 
 246:         /* 映射到一维数组 */
 247:         if (curr_y >= 0 && curr_y < MT9V03X_H) {
 248:             if (left_line_list[curr_y] == 255) {
 249:                 left_line_list[curr_y] = Limit_uint8(1, curr_x, MT9V03X_W - 2);
 250:             }
 251:         }
 252: 
 253:         /* 只有到达图像顶端才结束爬行 */
 254:         if (curr_y <= search_end_line) break;
 255: 
 256:         /* 寻找下一个边缘点 (向左转90度开始，扫6个方向防止倒退) */
 257:         found = 0;
 258:         start_search = (curr_dir + 6) % 8; 
 259:         
 260:         for (i = 0; i < 6; i++) {
 261:             search_dir = (start_search + i) % 8; /* 顺时针扫 */
 262:             nx = curr_x + dir_x[search_dir];
 263:             ny = curr_y + dir_y[search_dir];
 264: 
 265:             /* 允许贴着屏幕边缘走，不触发 break */
 266:             if (nx >= 1 && nx <= MT9V03X_W - 2 && ny >= search_end_line && ny < MT9V03X_H) {
 267:                 if (IS_WHITE(ny, nx)) {
 268:                     curr_x = nx;
 269:                     curr_y = ny;
 270:                     curr_dir = search_dir;
 271:                     found = 1;
 272:                     break;
 273:                 }
 274:             }
 275:         }
 276:         
 277:         /* 【防呆机制】如果进入纯黑死胡同，强制向上走一步，防止死循环 */
 278:         if (!found) {
 279:             curr_y--;
 280:             curr_dir = 0;
 281:         }
 282:     }
 283: 
 284:     /* ---------------- 右边缘八邻域爬行 ---------------- */
 285:     curr_x = start_right_x;
 286:     curr_y = start_y;
 287:     curr_dir = 0; /* 初始方向：向上 */
 288: 
 289:     while (right_pts_cnt < MT9V03X_H * 3) {
 290:         right_pts_x[right_pts_cnt] = curr_x;
 291:         right_pts_y[right_pts_cnt] = curr_y;
 292:         right_pts_dir[right_pts_cnt] = curr_dir;
 293:         right_pts_cnt++;
 294: 
 295:         if (curr_y >= 0 && curr_y < MT9V03X_H) {
 296:             if (right_line_list[curr_y] == 255) {
 297:                 right_line_list[curr_y] = Limit_uint8(1, curr_x, MT9V03X_W - 2);
 298:             }
 299:         }
 300: 
 301:         if (curr_y <= search_end_line) break;
 302: 
 303:         found = 0;
 304:         start_search = (curr_dir + 2) % 8; /* 向右转90度开始 */
 305:         
 306:         for (i = 0; i < 6; i++) {
 307:             search_dir = (start_search + 8 - i) % 8; /* 逆时针扫 */
 308:             nx = curr_x + dir_x[search_dir];
 309:             ny = curr_y + dir_y[search_dir];
 310: 
 311:             if (nx >= 1 && nx <= MT9V03X_W - 2 && ny >= search_end_line && ny < MT9V03X_H) {
 312:                 if (IS_WHITE(ny, nx)) {
 313:                     curr_x = nx;
 314:                     curr_y = ny;
 315:                     curr_dir = search_dir;
 316:                     found = 1;
 317:                     break;
 318:                 }
 319:             }
 320:         }
 321:         
 322:         if (!found) {
 323:             curr_y--;
 324:             curr_dir = 0;
 325:         }
 326:     }
 327: 
 328:     /* ---------------- 赛道空缺填补与中线计算 ---------------- */
 329:     for (i = search_start_line - 1; i > search_end_line; i--) {
 330:         if (left_line_list[i] == 255) {
 331:             left_line_list[i] = (i < search_start_line - 1) ? left_line_list[i + 1] : 1;
 332:         }
 333:         if (right_line_list[i] == 255) {
 334:             right_line_list[i] = (i < search_start_line - 1) ? right_line_list[i + 1] : MT9V03X_W - 2;
 335:         }
 336:         
 337:         left_line_raw[i] = left_line_list[i];
 338:         mid_line_list[i] = Limit_uint8(1, (left_line_list[i] + right_line_list[i]) / 2, MT9V03X_W - 2);
 339:     }
 340: }

 342: // ==========================================
 343: // 环岛方向预判 (丢线统计 + 中线绝对领地防伪)
 344: // ==========================================
 345: void judge_roundabout_dir(void)
 346: {
 347:     int r;
 348:     uint16 left_lost_cnt = 0;
 349:     uint16 right_lost_cnt = 0;
 350:     
 351:     // 获取屏幕中线坐标 (假设宽度是188，中线就是94)
 352:     int half_w = MT9V03X_W / 2; 
 353: 
 354:     if (roundabout_state > 0) return;
 355: 
 356:     // 1. 扫描中上部区域，统计左右两侧的“丢线（贴边）”行数
 357:     for (r = 30; r < MT9V03X_H - 15; r++) {
 358:         if (left_line_list[r] <= 5) {
 359:             left_lost_cnt++;
 360:         }
 361:         if (right_line_list[r] >= MT9V03X_W - 6) { 
 362:             right_lost_cnt++;
 363:         }
 364:     }
 365: 
 366:     island_dir = 0; // 默认为无环岛
 367: 
 368:     /* ==========================================
 369:      * 【判断左环岛】
 370:      * 基础条件：左边大面积丢线，且比右边多。
 371:      * 防伪条件：取画面中上部(Y=50)作为参考。右边线必须坚守在右半区！
 372:      * ========================================== */
 373:     if (left_lost_cnt > 15 && left_lost_cnt > right_lost_cnt + 10) {
 374:         
 375:         // 【核心绝杀】：如果是在左急弯，右边线在 Y=50 处一定会越过中线跑到左边。
 376:         // 我们给一点容错 (+10)，只要它越过了屏幕中线附近，就识破它是弯道！
 377:         if (right_line_list[50] < half_w + 10) {
 378:             island_dir = 0; // 越界了，这是个左转弯！
 379:         } 
 380:         else {
 381:             island_dir = 1; // 右线坚守右半区，确认前方是左环岛！
 382:         }
 383:     }
 384:     
 385:     /* ==========================================
 386:      * 【判断右环岛】(同理，镜像处理)
 387:      * ========================================== */
 388:     else if (right_lost_cnt > 15 && right_lost_cnt > left_lost_cnt + 10) {
 389:         
 390:         // 如果是右急弯，左边线在 Y=50 处会越过中线跑到右边。
 391:         // 给一点容错 (-10)，只要左线跑到了右半区，识破为弯道。
 392:         if (left_line_list[50] > half_w - 10) {
 393:             island_dir = 0; // 越界了，这是个右转弯！
 394:         } 
 395:         else {
 396:             island_dir = 2; // 左线坚守左半区，确认前方是右环岛！
 397:         }
 398:     }
 399: }

1002:             slope = (float)((int)pt_down_x - (int)pt_mid_x) / (float)((int)pt_down_y - (int)pt_mid_y);
1003:             for (r = pt_down_y; r >= (int)pt_mid_y; r--) {

1010: 
1011:     // --- 状态 2：A 消失，从 B 点直接连到屏幕左下角 ---
1012:     else if (roundabout_state == R_A_LOST) {
1013:         if (pt_mid_x > 0 && pt_mid_y < MT9V03X_H - 1) { 
1014:             slope = (float)(1 - (int)pt_mid_x) / (float)((MT9V03X_H - 1) - (int)pt_mid_y);
1015:             for (r = pt_mid_y; r < MT9V03X_H; r++) {
1016:                 new_x = (int)pt_mid_x + (int)(slope * (float)(r - (int)pt_mid_y));
1017:                 left_line_list[r] = Limit_uint8(1, new_x, MT9V03X_W - 2);

1019:             }
1020:         }
1021:     }
